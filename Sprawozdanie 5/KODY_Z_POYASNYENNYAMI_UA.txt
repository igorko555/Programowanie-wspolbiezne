using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.Threading;

namespace UnsafeCollection
{
    class Punkt2
    {
        static void Main(string[] args)
        {
            // Створюємо список потоків для подальшої синхронізації через Join()
            List<Thread> watki = new List<Thread>();

            // Звичайна колекція List<T> НЕ є потокобезпечною. Вона інкапсулює внутрішній масив T[].
            List<int> liczby = new List<int>(1000000);
            Random rand = new Random();

            // Вимірювання часу виконання за допомогою Stopwatch
            Stopwatch sw = Stopwatch.StartNew();

            // Запускаємо 100 потоків, кожен з яких намагається додати 10 000 елементів
            for (int i = 0; i < 100; i++)
            {
                var watek = new Thread(() =>
                {
                    for (int l = 0; l < 10000; l++)
                    {
                        try
                        {
                            // Одночасний виклик Add() без блокувань призводить до перезапису однакових індексів
                            // та виходу за межі масиву під час його динамічного розширення
                            liczby.Add(rand.Next());
                        }
                        catch
                        {
                            // Перехоплення винятків пошкодження внутрішньої структури списку
                        }
                    }
                });
                watki.Add(watek);
                watek.Start();
            }

            // Очікуємо завершення роботи всіх 100 потоків
            foreach (var watek in watki)
            {
                watek.Join();
            }

            sw.Stop();
            Console.WriteLine($"Czas wykonania: {sw.ElapsedMilliseconds} ms");
            Console.WriteLine($"Liczba elementow w liscie zwyklej: {liczby.Count} (oczekiwano: 1000000)");
        }
    }
}

    {

        {
            List<Thread> watki = new List<Thread>();

            // ConcurrentBag<T> реалізує внутрішню структуру локальних черг для кожного потоку (ThreadLocalList).
            // Додавання елементів виконується майже без блокувань за алгоритмом Work Stealing.

                        liczby.Add(rand.Next());
                    }
                });
                watki.Add(watek);
                watek.Start();
            }

            foreach (var watek in watki)
            {
                watek.Join();
            }

            sw.Stop();
            Console.WriteLine($"Czas wykonania (ConcurrentBag): {sw.ElapsedMilliseconds} ms");

        }
    }

using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.Threading;

namespace UnsafeCollection
{
    class Punkt4
    {
        // Об'єкт синхронізації для створення секції взаємного виключення (mutual exclusion)
        static readonly object locker = new object();

        static void Main(string[] args)
        {
            List<Thread> watki = new List<Thread>();
            List<int> liczby = new List<int>(1000000);
            Stopwatch sw = Stopwatch.StartNew();

            for (int i = 0; i < 100; i++)
            {
                var watek = new Thread(() =>
                {
                    Random rand = new Random();
                    for (int l = 0; l < 10000; l++)
                    {
                        int val = rand.Next();
                        // lock гарантує, що лише один потік у певний момент часу виконує операцію Add()
                        lock (locker)
                        {
                            liczby.Add(val);
                        }
                    }
                });
                watki.Add(watek);
                watek.Start();
            }

            foreach (var watek in watki)
            {
                watek.Join();
            }

            sw.Stop();
            Console.WriteLine($"Czas wykonania (List + lock): {sw.ElapsedMilliseconds} ms");
            Console.WriteLine($"Liczba elementow w liscie z lock: {liczby.Count} (oczekiwano: 1000000)");
        }
    }
}

using System;
using System.Collections.Concurrent;
using System.Collections.Generic;
using System.Diagnostics;
using System.Linq;
using System.Threading;

namespace UniqueCollection
{
    class Punkt5
    {
        static readonly object locker = new object();

        static void Main(string[] args)
        {
            List<Thread> watki = new List<Thread>();
            ConcurrentBag<int> liczby = new ConcurrentBag<int>();
            Stopwatch sw = Stopwatch.StartNew();

            for (int i = 0; i < 100; i++)
            {
                var watek = new Thread(() =>
                {
                    Random rand = new Random();
                    for (int l = 0; l < 100; l++)
                    {
                        int liczba = rand.Next(10001);

                        // Усунення стану гонки TOCTOU (Time-of-Check to Time-of-Use):
                        // Перевірка Any() та додавання Add() поміщені в неподільну секцію lock
                        lock (locker)
                        {
                            while (liczby.Any(x => x == liczba))
                            {
                                liczba = rand.Next(10001);
                            }
                            liczby.Add(liczba);
                        }
                    }
                });
                watki.Add(watek);
                watek.Start();
            }

            foreach (var watek in watki)
            {
                watek.Join();
            }

            sw.Stop();
            Console.WriteLine($"Uplynelo {sw.ElapsedMilliseconds} ms");
            Console.WriteLine($"Liczba elementow: {liczby.Count}");

            // Перевірка розширювальним методом IsUnique()
            if (liczby.IsUnique())
            {
                Console.WriteLine("wszystkie liczby sa unikalne");
            }
            else
            {
                Console.WriteLine("Nie wszystkie liczby sa unikalne");
            }
        }
    }
}

using System;
using System.Collections.Concurrent;
using System.Collections.Generic;
using System.Diagnostics;
using System.Threading;

namespace UniqueCollection
{
    class Punkt6
    {
        static void Main(string[] args)
        {
            List<Thread> watki = new List<Thread>();

            // ConcurrentDictionary гарантує унікальність ключів на рівні хеш-таблиці
            // за допомогою атомарного методу TryAdd() без потреби глобального lock
            ConcurrentDictionary<int, byte> dict = new ConcurrentDictionary<int, byte>();
            Stopwatch sw = Stopwatch.StartNew();

            for (int i = 0; i < 100; i++)
            {
                var watek = new Thread(() =>
                {
                    Random rand = new Random();
                    for (int l = 0; l < 100; l++)
                    {
                        int liczba = rand.Next(100000);
                        // Якщо число вже існує в таблиці, TryAdd повертає false, і ми генеруємо нове
                        while (!dict.TryAdd(liczba, 0))
                        {
                            liczba = rand.Next(100000);
                        }
                    }
                });
                watki.Add(watek);
                watek.Start();
            }

            foreach (var watek in watki)
            {
                watek.Join();
            }

            sw.Stop();
            Console.WriteLine($"Uplynelo {sw.ElapsedMilliseconds} ms");
            Console.WriteLine($"Liczba unikalnych elementow w ConcurrentDictionary: {dict.Count} (oczekiwano: 10000)");
            Console.WriteLine("Wszystkie elementy w slowniku sa unikalne z definicji klucza.");
        }
    }
}

using System;
using System.Collections.Concurrent;
using System.Collections.Generic;
using System.Threading;

namespace ProducerConsumerQ
{
    class Punkt7
    {
        static void Main(string[] args)
        {
            ConcurrentQueue<int> cq = new ConcurrentQueue<int>();

            // SemaphoreSlim(0) інкапсулює кількість доступних для читання елементів.
            // Якщо черга порожня, лічильник = 0, і споживачі переходять у стан блокування
            SemaphoreSlim sem = new SemaphoreSlim(0);
            List<Thread> watki = new List<Thread>();
            Random rand = new Random();

            var producent = new Thread(() =>
            {
                for (int l = 0; l < 100; l++)
                {
                    cq.Enqueue(l);
                    Console.WriteLine($"Wkladam {l}");
                    // Збільшуємо лічильник семафора, щоб розбудити одного очікуючого споживача
                    sem.Release();
                    Thread.Sleep(rand.Next(10));
                }
            });
            producent.Start();

            // 10 споживачів, кожен зчитує рівно по 10 елементів (разом 100)
            for (int i = 0; i < 10; i++)
            {
                var watek = new Thread(() =>
                {
                    for (int l = 0; l < 10; l++)
                    {
                        // Очікуємо сигналу від виробника (без марного навантаження CPU)
                        sem.Wait();
                        if (cq.TryDequeue(out int result2))
                        {
                            Console.WriteLine($"Odbieram {result2}");
                        }
                        Thread.Sleep(rand.Next(20));
                    }
                });
                watki.Add(watek);
                watek.Start();
            }

            foreach (var watek in watki)
            {
                watek.Join();
            }
            producent.Join();
            Console.WriteLine("Wszystkie elementy zostaly poprawnie odebrane.");
        }
    }
}

using System;
using System.Collections.Concurrent;
using System.Collections.Generic;
using System.Threading;

namespace BlockingCollectionDemo
{
    class Punkt8
    {
        static void Main(string[] args)
        {
            // Обмежений буфер ємністю 10 елементів.
            // Якщо буфер заповнений, метод Add() автоматично блокує виробника!
            BlockingCollection<int> buffer = new BlockingCollection<int>(10);
            List<Thread> watki = new List<Thread>();

            var producent = new Thread(() =>
            {
                for (int l = 0; l < 100; l++)
                {
                    buffer.Add(l);
                    Console.WriteLine($"Producent wstawia: {l}");
                    Thread.Sleep(50);
                }
                // Сигналізуємо, що додавання завершено
                buffer.CompleteAdding();
            });
            producent.Start();

            // За вимогою інструкції: затримка 5 секунд перед стартом споживачів
            Thread.Sleep(5000);

            for (int i = 0; i < 10; i++)
            {
                int id = i;
                var watek = new Thread(() =>
                {
                    Random rand = new Random();
                    for (int l = 0; l < 10; l++)
                    {
                        try
                        {
                            // Take() автоматично блокує потік, якщо буфер порожній
                            int item = buffer.Take();
                            Console.WriteLine($"Konsument [{id}] pobral: {item}");
                        }
                        catch (InvalidOperationException)
                        {
                            // Буфер вичерпано і CompleteAdding викликано
                            break;
                        }
                        Thread.Sleep(rand.Next(500));
                    }
                });
                watki.Add(watek);
                watek.Start();
            }

            foreach (var watek in watki)
            {
                watek.Join();
            }
            producent.Join();
            Console.WriteLine("Praca bufora zakonczona.");
        }
    }
}

1. Чому звичайна List<T> руйнується при паралельному додаванні елементів, а ConcurrentBag<T> працює надійно?
ВІДПОВІДЬ:
- List<T> інкапсулює звичайний одномірний масив T[] items та цілочисельний лічильник int size.
- Коли кілька потоків одночасно викликають метод Add(), відбувається одночасний запис за одним і тим самим індексом, а також паралельне збільшення ємності (EnsureCapacity), через що один потік може перезаписати значення іншого, або посилатися на старий масив під час його реаллокації.
- ConcurrentBag<T> реалізує концепцію локальних списків для кожного потоку (ThreadLocalList). Коли потік додає елемент у колекцію, він поміщає його у свою власну голову двозв'язного списку без глобального блокування. Якщо потоку потрібно забрати елемент, він спочатку бере свій власний елемент (LIFO), і лише якщо його список порожній, намагається «вкрасти» (Work Stealing) елемент із хвоста іншого потоку через коротку атомарну синхронізацію.

2. У чому полягає помилка TOCTOU (Time-of-Check to Time-of-Use) у завданні з унікальною колекцією?
ВІДПОВІДЬ:
- Класична помилка виникає, коли між операцією перевірки умови (Check) та операцією виконання дії (Act / Use) виникає часовий інтервал, під час якого інший потік встигає змінити стан системи.
- У коді з ConcurrentBag:
    while (numbers.Any(x => x == number)) { number = rand.Next(); }
    numbers.Add(number);
  Потік А перевіряє колекцію: числа ще немає. Потім планувальник ОС перемикає контекст на Потік Б. Потік Б теж перевіряє — числа ще немає, і додає його. Потім продовжує роботу Потік А і теж додає це саме число! В результаті в колекції з'являються дублікати.
- Виправлення 1: Захистити блок перевірки і додавання глобальним ryglem lock.
- Виправлення 2: Використати атомарний метод TryAdd() колекції ConcurrentDictionary<int, byte>, який виконує перевірку наявності ключа та його вставку як одну неподільну операцію.

3. Чому використання SemaphoreSlim у ProducerConsumerQ набагато ефективніше за безкінечний цикл із TryDequeue()?
ВІДПОВІДЬ:
- Якщо потік-споживач у циклі викликає TryDequeue(), і якщо черга порожня, він продовжує молотити процесорні такти (активне очікування / busy waiting, або SpinWait), навантажуючи CPU на 100%.
- SemaphoreSlim інкапсулює лічильник доступних елементів черги. Споживач викликає sem.Wait(), і якщо лічильник рівний 0, операційна система негайно переводить потік у стан сну (WaitSleepJoin), звільняючи ядро процесора для інших задач. Коли виробник додає елемент (Enqueue), він викликає sem.Release(), і тільки тоді ОС миттєво будить один очікуючий потік-споживач.

4. Які переваги надає BlockingCollection<T> для реалізації патерну Producent-Konsument?
ВІДПОВІДЬ:
- BlockingCollection<T> забезпечує обмежену ємність (bounded capacity). Якщо буфер заповнений (наприклад, 10 елементів), виклик Add() автоматично блокує потік виробника, запобігаючи переповненню оперативної пам'яті (OutOfMemoryException).
- Метод Take() автоматично блокує споживача, якщо буфер порожній, повністю усуваючи активне очікування.
- Метод CompleteAdding() безпечно сповіщає всіх споживачів, що нових даних більше не буде, дозволяючи їм вичитати залишки і штатно завершити цикл foreach (var item in buffer.GetConsumingEnumerable()).
