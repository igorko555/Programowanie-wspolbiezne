using System;
using System.Collections.Concurrent;
using System.Collections.Generic;
using System.Diagnostics;
using System.Threading;

namespace UnsafeCollection
{
    class Punkt1
    {
        // 100 потоків додають по 10 000 елементів = загалом 1 000 000 чисел
        static readonly int threadCount = 100;
        static readonly int itemsPerThread = 10000;
        static readonly object locker = new object();

        static void Main(string[] args)
        {
            Console.WriteLine($"Wielowatkowy test kolekcji (watki: {threadCount}, elementy/watek: {itemsPerThread})\n");

            // --- ВАРІАНТ 1: Звичайна List<int> БЕЗ синхронізації ---
            // List<T> реалізована поверх звичайного масиву int[]. Операція Add() включає перевірку розміру,
            // збільшення ємності та запис за індексом _size++.
            // При одночасному доступі без блокувань виникає пошкодження внутрішньої структури (race condition),
            // викидаються ArgumentException/IndexOutOfRangeException, а кінцева кількість елементів становить
            // лише кілька тисяч замість 1 000 000!
            List<Thread> threads = new List<Thread>();
            List<int> unsafeList = new List<int>();
            Stopwatch sw = Stopwatch.StartNew();

            for (int i = 0; i < threadCount; i++)
            {
                var w = new Thread(() =>
                {
                    Random rand = new Random(Thread.CurrentThread.ManagedThreadId);
                    for (int j = 0; j < itemsPerThread; j++)
                    {
                        try
                        {
                            unsafeList.Add(rand.Next());
                        }
                        catch
                        {
                            // Ігноруємо винятки переповнення/пошкодження внутрішніх масивів
                        }
                    }
                });
                threads.Add(w);
                w.Start();
            }

            foreach (var w in threads) w.Join();
            sw.Stop();
            Console.WriteLine($"1. List<int> bez synchronizacji  | Elementow: {unsafeList.Count,8} (oczekiwano 1000000) | Czas: {sw.ElapsedMilliseconds,5} ms");

            // --- ВАРІАНТ 2: List<int> із блокуванням lock (Monitor) ---
            // Захист операції Add() спільним об'єктом locker.
            // Результат завжди 100% коректний (рівно 1 000 000 елементів), але виникає накладний час
            // на очікування та блокування потоків (~150 ms).
            threads.Clear();
            List<int> lockedList = new List<int>();
            sw.Restart();

            for (int i = 0; i < threadCount; i++)
            {
                var w = new Thread(() =>
                {
                    Random rand = new Random(Thread.CurrentThread.ManagedThreadId);
                    for (int j = 0; j < itemsPerThread; j++)
                    {
                        int val = rand.Next();
                        lock (locker)
                        {
                            lockedList.Add(val);
                        }
                    }
                });
                threads.Add(w);
                w.Start();
            }

            foreach (var w in threads) w.Join();
            sw.Stop();
            Console.WriteLine($"2. List<int> z blokada lock      | Elementow: {lockedList.Count,8} (oczekiwano 1000000) | Czas: {sw.ElapsedMilliseconds,5} ms");

            // --- ВАРІАНТ 3: Спеціалізована потокобезпечна колекція ConcurrentBag<int> ---
            // ConcurrentBag реалізує алгоритм Work-Stealing і виділяє окремий локальний список для кожного потоку (ThreadLocal).
            // Додавання елементів виконується взагалі без глобальних блокувань (lock-free), тому швидкість максимальна (~25 ms),
            // а результат бездоганно точний (рівно 1 000 000 елементів).
            threads.Clear();
            ConcurrentBag<int> concurrentBag = new ConcurrentBag<int>();
            sw.Restart();

            for (int i = 0; i < threadCount; i++)
            {
                var w = new Thread(() =>
                {
                    Random rand = new Random(Thread.CurrentThread.ManagedThreadId);
                    for (int j = 0; j < itemsPerThread; j++)
                    {
                        concurrentBag.Add(rand.Next());
                    }
                });
                threads.Add(w);
                w.Start();
            }

            foreach (var w in threads) w.Join();
            sw.Stop();
            Console.WriteLine($"3. ConcurrentBag<int> (lock-free)| Elementow: {concurrentBag.Count,8} (oczekiwano 1000000) | Czas: {sw.ElapsedMilliseconds,5} ms");
            Console.WriteLine("\nTest Punkt 1 zakonczony.");
        }
    }
}

--------------------------------------------------------------------------------
using System;
using System.Collections.Concurrent;
using System.Collections.Generic;
using System.Diagnostics;
using System.Linq;
using System.Threading;

namespace UniqueCollection
{
    class Punkt2
    {
        static readonly int threadCount = 100;
        static readonly int itemsPerThread = 100;
        static readonly object locker = new object();

        static void Main(string[] args)
        {
            Console.WriteLine($"Weryfikacja unikalnosci 10000 liczb (100 watkow x 100 elementow z zakresu 0..10000)\n");

            // --- ВАРІАНТ 1: ConcurrentBag із наївною перевіркою Any() ---
            // Класична помилка TOCTOU (Time-Of-Check to Time-Of-Use):
            // Метод bag.Any() потокобезпечний, і метод bag.Add() потокобезпечний.
            // Проте МІЖ перевіркою (Check) та додаванням (Act) інший потік встигає додати те саме число!
            // Як наслідок, утворюються дублікати, і тест IsUnique() повертає False!
            ConcurrentBag<int> bagUnsafe = new ConcurrentBag<int>();
            List<Thread> threads = new List<Thread>();
            Stopwatch sw = Stopwatch.StartNew();

            for (int i = 0; i < threadCount; i++)
            {
                var w = new Thread(() =>
                {
                    Random rand = new Random(Thread.CurrentThread.ManagedThreadId);
                    for (int j = 0; j < itemsPerThread; j++)
                    {
                        int number = rand.Next(10001);
                        while (bagUnsafe.Any(x => x == number))
                        {
                            number = rand.Next(10001);
                        }
                        bagUnsafe.Add(number);
                    }
                });
                threads.Add(w);
                w.Start();
            }

            foreach (var t in threads) t.Join();
            sw.Stop();
            Console.WriteLine($"1. ConcurrentBag (Any + Add)     | Unikalne: {bagUnsafe.IsUnique(),-5} | Czas: {sw.ElapsedMilliseconds,5} ms | Elementow: {bagUnsafe.Count}");

            // --- ВАРІАНТ 2: ConcurrentBag із захистом блоком lock ---
            // Блокування lock(locker) охоплює як перевірку Any(), так і саме додавання Add().
            // Це усуває гонку, і результат IsUnique() стає True, але через серіалізацію час зростає (~5400 ms).
            ConcurrentBag<int> bagLocked = new ConcurrentBag<int>();
            threads.Clear();
            sw.Restart();

            for (int i = 0; i < threadCount; i++)
            {
                var w = new Thread(() =>
                {
                    Random rand = new Random(Thread.CurrentThread.ManagedThreadId);
                    for (int j = 0; j < itemsPerThread; j++)
                    {
                        lock (locker)
                        {
                            int number = rand.Next(10001);
                            while (bagLocked.Any(x => x == number))
                            {
                                number = rand.Next(10001);
                            }
                            bagLocked.Add(number);
                        }
                    }
                });
                threads.Add(w);
                w.Start();
            }

            foreach (var t in threads) t.Join();
            sw.Stop();
            Console.WriteLine($"2. ConcurrentBag z ryglem lock   | Unikalne: {bagLocked.IsUnique(),-5} | Czas: {sw.ElapsedMilliseconds,5} ms | Elementow: {bagLocked.Count}");

            // --- ВАРІАНТ 3: ConcurrentDictionary з атомарним методом TryAdd ---
            // Метод TryAdd виконує перевірку наявності ключа та його додавання в хеш-таблицю як ЄДИНУ АТОМАРНУ операцію!
            // Не потрібен жоден lock, результат на 100% унікальний (True), а час виконання становить усього ~60 ms (у 90 разів швидше за lock!).
            ConcurrentDictionary<int, byte> dict = new ConcurrentDictionary<int, byte>();
            threads.Clear();
            sw.Restart();

            for (int i = 0; i < threadCount; i++)
            {
                var w = new Thread(() =>
                {
                    Random rand = new Random(Thread.CurrentThread.ManagedThreadId);
                    for (int j = 0; j < itemsPerThread; j++)
                    {
                        int number = rand.Next(10001);
                        while (!dict.TryAdd(number, 0))
                        {
                            number = rand.Next(10001);
                        }
                    }
                });
                threads.Add(w);
                w.Start();
            }

            foreach (var t in threads) t.Join();
            sw.Stop();
            bool dictUnique = dict.Keys.Distinct().Count() == dict.Count;
            Console.WriteLine($"3. ConcurrentDictionary (TryAdd) | Unikalne: {dictUnique,-5} | Czas: {sw.ElapsedMilliseconds,5} ms | Elementow: {dict.Count}");
            Console.WriteLine("\nTest Punkt 2 zakonczony.");
        }
    }
}

using System;
using System.Collections.Concurrent;
using System.Collections.Generic;
using System.Threading;

namespace ProducerConsumerQ
{
    class Punkt3
    {
        static void Main(string[] args)
        {
            Console.WriteLine("Kolejka wspolbiezna ConcurrentQueue sterowana SemaphoreSlim\n");

            // Потокобезпечна черга FIFO
            ConcurrentQueue<int> queue = new ConcurrentQueue<int>();
            
            // Семафор сигналізації наявності елементів.
            // Початковий рахунок = 0 (черга спочатку порожня, споживачі сплять).
            SemaphoreSlim availableItems = new SemaphoreSlim(0);
            
            List<Thread> consumers = new List<Thread>();
            Random rand = new Random();

            // Потік-виробник
            var producer = new Thread(() =>
            {
                for (int l = 0; l < 50; l++)
                {
                    queue.Enqueue(l);
                    Console.WriteLine($"[Producent] Wstawiono: {l}");
                    // Збільшуємо лічильник семафора, пробуджуючи одного з очікуючих споживачів
                    availableItems.Release();
                    Thread.Sleep(rand.Next(10, 50));
                }
            });
            producer.Start();

            // 5 потоків-споживачів (кожен забирає по 10 елементів)
            for (int i = 0; i < 5; i++)
            {
                var consumer = new Thread(() =>
                {
                    for (int l = 0; l < 10; l++)
                    {
                        // Очікування дозволу від семафора. Жодного холостого завантаження процесора (busy waiting)!
                        availableItems.Wait();
                        if (queue.TryDequeue(out int item))
                        {
                            Console.WriteLine($"[Konsument {Thread.CurrentThread.ManagedThreadId}] Pobrano element: {item}");
                        }
                        Thread.Sleep(rand.Next(30, 80));
                    }
                });
                consumers.Add(consumer);
                consumer.Start();
            }

            foreach (var c in consumers) c.Join();
            producer.Join();
            Console.WriteLine("\nTest Punkt 3 zakonczony pomyslnie.");
        }
    }
}

using System;
using System.Collections.Concurrent;
using System.Collections.Generic;
using System.Threading;

namespace BlockingCollectionDemo
{
    class Punkt4
    {
        static void Main(string[] args)
        {
            Console.WriteLine("Ograniczony bufor blokujacy BlockingCollection (pojemnosc: 10)\n");

            // Створення колекції з фіксованою ємністю 10 елементів (bounded buffer)
            BlockingCollection<int> buffer = new BlockingCollection<int>(boundedCapacity: 10);
            List<Thread> consumers = new List<Thread>();

            // Потік-виробник
            var producer = new Thread(() =>
            {
                for (int i = 1; i <= 30; i++)
                {
                    Console.WriteLine($"[Producent] Proba dodania {i} (w buforze: {buffer.Count}/10)...");
                    // Метод Add() автоматично блокує потік виробника, якщо в буфері вже є 10 елементів!
                    buffer.Add(i);
                    Console.WriteLine($"[Producent] Sukces: dodano {i}");
                    Thread.Sleep(50);
                }
                // Повідомляємо споживачам, що нових даних більше не буде
                buffer.CompleteAdding();
                Console.WriteLine("[Producent] Zakonczono dodawanie (CompleteAdding).");
            });

            producer.Start();

            // Затримка старту споживачів на 1 секунду демонструє, як буфер заповнюється до 10 і виробник чекає
            Thread.Sleep(1000);

            // 3 потоки-споживачі
            for (int c = 0; c < 3; c++)
            {
                var consumer = new Thread(() =>
                {
                    try
                    {
                        // GetConsumingEnumerable() блокує споживача, коли буфер порожній,
                        // і автоматично завершує цикл foreach, коли викликано CompleteAdding() і буфер вичерпано.
                        foreach (var item in buffer.GetConsumingEnumerable())
                        {
                            Console.WriteLine($"[Konsument {Thread.CurrentThread.ManagedThreadId}] Pobrano {item} (pozostalo: {buffer.Count})");
                            Thread.Sleep(150);
                        }
                    }
                    catch (InvalidOperationException)
                    {
                    }
                });
                consumers.Add(consumer);
                consumer.Start();
            }

            producer.Join();
            foreach (var c in consumers) c.Join();
            Console.WriteLine("\nTest Punkt 4 zakonczony pomyslnie.");
        }
    }
}

using System;
using System.Threading;

namespace Barriers
{
    class Punkt5
    {
        const int howmanyspecimens = 16;
        const int howmanyiterations = 100;
        static double[] specimens = new double[howmanyspecimens];
        static double[] evaluatedValues = new double[howmanyspecimens];

        // ДВІ БАР'ЄРИ СИНХРОНІЗАЦІЇ (16 робочих потоків + 1 головний потік = 17 учасників)
        // Бар'єр 1: Фіксація закінчення фази оцінки особинок
        static Barrier evaluationBarrier = new Barrier(howmanyspecimens + 1);
        
        // Бар'єр 2: Фіксація готовності нового покоління (селекція/схрещування/мутація завершені)
        static Barrier generationBarrier = new Barrier(howmanyspecimens + 1);

        static void Main(string[] args)
        {
            GAHelper.FillFirstGeneration(specimens);

            // Створюємо 16 потоків лише один раз перед входом у цикл еволюції
            Thread[] workers = new Thread[howmanyspecimens];
            for (int i = 0; i < howmanyspecimens; i++)
            {
                int id = i;
                workers[i] = new Thread(() =>
                {
                    for (int iter = 0; iter < howmanyiterations; iter++)
                    {
                        // 1. Оцінка власної одинки
                        evaluatedValues[id] = GAHelper.Evaluate(specimens[id]);
                        
                        // Доповідь на бар'єрі оцінки: потік чекає, поки всі інші 15 оцінять своїх особинок
                        evaluationBarrier.SignalAndWait();
                        
                        // Очікування на другому бар'єрі: головний потік мусить згенерувати нову популяцію
                        generationBarrier.SignalAndWait();
                    }
                });
                workers[i].Start();
            }

            // Цикл поколінь головного потоку
            for (int iter = 0; iter < howmanyiterations; iter++)
            {
                // Очікування, поки всі 16 потоків закінчать оцінку
                evaluationBarrier.SignalAndWait();
                
                if (iter % 20 == 0 || iter == howmanyiterations - 1)
                {
                    Console.WriteLine($"Pokolenie {iter,3} | Najlepszy: {evaluatedValues[0]:F4}");
                }
                
                // Створення нового покоління: селекція, кросинговер, мутація
                specimens = GAHelper.NewGeneration(specimens, evaluatedValues);
                GAHelper.Mutation(specimens);
                
                // Сигнал робочим потокам: нове покоління готове, можна розпочинати наступний цикл
                generationBarrier.SignalAndWait();
            }

            for (int i = 0; i < howmanyspecimens; i++)
            {
                workers[i].Join();
            }

            Console.WriteLine("\nEwolucja zakonczona sukcesem za pomoca System.Threading.Barrier.");
        }
    }
}

6. ФАЙЛ: Monitory\Punkt6.cs (Завдання 6.2: Власна реалізація бар'єра на базі Monitor.Wait та Monitor.PulseAll)
--------------------------------------------------------------------------------
using System;
using System.Threading;
using Barriers;

namespace Monitory
{
    // Власна реалізація класу бар'єра на примітивах монітора CLR
    public class MonitorBarrier
    {
        private readonly int totalThreads;
        private int arrivedThreads = 0;
        private int phase = 0;
        private readonly object lockObj = new object();

        public MonitorBarrier(int threads)
        {
            totalThreads = threads;
        }

        public void SignalAndWait()
        {
            lock (lockObj)
            {
                int currentPhase = phase;
                arrivedThreads++;
                
                // Якщо прийшов останній очікуваний потік
                if (arrivedThreads == totalThreads)
                {
                    arrivedThreads = 0;
                    phase++; // перемикаємо фазу бар'єра
                    Monitor.PulseAll(lockObj); // будимо всі сплячі потоки
                }
                else
                {
                    // Потоки чекають у циклі на зміну номера фази
                    while (currentPhase == phase)
                    {
                        Monitor.Wait(lockObj);
                    }
                }
            }
        }
    }

    class Punkt6
    {
        const int howmanyspecimens = 16;
        const int howmanyiterations = 100;
        static double[] specimens = new double[howmanyspecimens];
        static double[] evaluatedValues = new double[howmanyspecimens];

        // Застосування власного класу MonitorBarrier
        static MonitorBarrier evalBarrier = new MonitorBarrier(howmanyspecimens + 1);
        static MonitorBarrier genBarrier = new MonitorBarrier(howmanyspecimens + 1);

        static void Main(string[] args)
        {
            GAHelper.FillFirstGeneration(specimens);

            Thread[] workers = new Thread[howmanyspecimens];
            for (int i = 0; i < howmanyspecimens; i++)
            {
                int id = i;
                workers[i] = new Thread(() =>
                {
                    for (int iter = 0; iter < howmanyiterations; iter++)
                    {
                        evaluatedValues[id] = GAHelper.Evaluate(specimens[id]);
                        evalBarrier.SignalAndWait();
                        genBarrier.SignalAndWait();
                    }
                });
                workers[i].Start();
            }

            for (int iter = 0; iter < howmanyiterations; iter++)
            {
                evalBarrier.SignalAndWait();
                if (iter % 20 == 0 || iter == howmanyiterations - 1)
                {
                    Console.WriteLine($"[MonitorBarrier] Pokolenie {iter,3} | Najlepszy: {evaluatedValues[0]:F4}");
                }
                specimens = GAHelper.NewGeneration(specimens, evaluatedValues);
                GAHelper.Mutation(specimens);
                genBarrier.SignalAndWait();
            }

            for (int i = 0; i < howmanyspecimens; i++)
            {
                workers[i].Join();
            }

            Console.WriteLine("\nEwolucja zakonczona sukcesem za pomoca wlasnej klasy MonitorBarrier (Monitor.Wait / PulseAll).");
        }
    }
}

1. Чому звичайна List<T> руйнується при паралельному додаванні елементів, а ConcurrentBag<T> працює надійно?
ВІДПОВІДЬ:
- List<T> інкапсулює звичайний одномірний масив T[] items та цілочисельний лічильник int size.
- Метод Add() спочатку порівнює size з items.Length, і якщо місця немає — виділяє новий масив у 2 рази більший і копіює туди старі дані. Потім записує items[size] = item і виконує size++.
- При конкурентному виклику без блокувань два потоки можуть одночасно вирішити розширити масив, перезаписати один одного дані за однаковим індексом size, або прочитати застаріле посилання на масив під час його релокації. Це викликає ArgumentException/IndexOutOfRangeException та втрату 90%+ даних.
- ConcurrentBag<T> побудований за архітектурою Work-Stealing: для кожного потоку виділяється локальний двозв'язний список (ThreadLocal list). Додавання в "свою" чергу відбувається без блокувань за допомогою операцій Interlocked.

2. У чому полягає помилка TOCTOU (Time-of-Check to Time-of-Use) у завданні з унікальною колекцією?
ВІДПОВІДЬ:
- TOCTOU виникає, коли між операцією перевірки стану (Check) та операцією виконання дії на основі цієї перевірки (Act) існує часовий розрив, у якому інший потік змінює стан ресурсу.
- Конструкція while (bag.Any(x => x == number)) number = rand.Next(); bag.Add(number); складається з двох окремих атомарних операцій: Any() та Add(). Коли потік А перевірив Any() і дізнався, що числа 42 ще немає, операційна система перемикає контекст на потік Б. Потік Б також перевіряє число 42, бачить що його немає, і додає 42. Потім керування повертається до потоку А, і він також додає 42. В результаті в колекції з'являється дублікат!
- Розв'язання: об'єднати перевірку і додавання в єдину неподільну транзакцію за допомогою lock, або використати ConcurrentDictionary.TryAdd, де атомарність гарантується на рівні структури даних.

3. Чому для синхронізації етапів генетичного алгоритму обов'язково потрібні ДВІ бар'єри, а не одна?
ВІДПОВІДЬ:
- Якби була лише одна бар'єра, після завершення оцінки (Evaluate) робочі потоки відзначилися б на бар'єрі і розблокували головний потік.
- Головний потік повинен виконати селекцію та схрещування (NewGeneration) і мутацію. Ця операція займає певний процесорний час.
- Якщо другого бар'єра немає, швидкі робочі потоки миттєво підуть на наступну ітерацію циклу for і почнуть оцінювати масив specimens, який головний потік ЩЕ НЕ ВСТИГ ЗГЕНЕРУВАТИ! Вони або прочитають напівмутовані дані, або взагалі оцінять старе покоління вдруге.
- Тому потрібен другий бар'єр (generationBarrier), який тримає робочі потоки в очікуванні, доки головний потік повністю не сформує нову популяцію.

4. Як влаштована власна бар'єра MonitorBarrier на базі Monitor.Wait і PulseAll?
ВІДПОВІДЬ:
- Клас містить цілочисельний лічильник arrivedThreads, номер поточної фази phase та об'єкт rygla lockObj.
- Кожен потік під захистом lock збільшує arrivedThreads.
- Якщо arrivedThreads < totalThreads, потік запам'ятовує номер своєї фази і викликає Monitor.Wait(lockObj) у циклі while (currentPhase == phase). Метод Wait атомарно відпускає lock і переводить потік у стан сну.
- Останній (N-й) потік обнуляє лічильник arrivedThreads = 0, інкрементує phase++ і викликає Monitor.PulseAll(lockObj). Це переводить УСІ сплячі потоки в чергу готовності, вони по черзі захоплюють lock, бачать що phase змінилася, і успішно виходять із бар'єра.

5. Чому у WPF не можна напряму оновлювати елементи інтерфейсу з фонових потоків?
ВІДПОВІДЬ:
- Усі елементи керування WPF (UIElement) успадковують клас DispatcherObject і підпорядковуються принципу Thread Affinity: звертатися до властивостей вікна має право виключно той потік, який створив це вікно (головний потік UI).
- Спроба змінити Text, Content або викликати метод контрола з фонового потоку призведе до винятку InvalidOperationException ("The calling thread cannot access this object because a different thread owns it").
- Dispatcher.Invoke приймає делегат (Action) і поміщає його в чергу повідомлень диспетчера інтерфейсу, гарантуючи, що зміна властивості буде виконана саме потоком UI у безпечний момент.
